1. What are operators?
Operators perform operations on variables and values. Example: +, -, *, ==, &&, etc.

2. Arithmetic operators with example:
Used to perform mathematical operations.

javascript

let a = 10, b = 3;
console.log(a + b); // 13
console.log(a - b); // 7
console.log(a * b); // 30
console.log(a / b); // 3.33
console.log(a % b); // 1
3. Logical operators with example:
Used to combine multiple conditions.

javascript

let x = 10, y = 20;
console.log(x > 5 && y > 15); // true
console.log(x > 15 || y > 15); // true
console.log(!(x > 5)); // false
4. Assignment operators with example:
Used to assign values to variables.

javascript

let a = 5;
a += 3; // a = a + 3
console.log(a); // 8
5. If statement with example:
javascript

let age = 18;
if (age >= 18) {
  console.log("Eligible to vote");
}
6. Switch statement with example:
javascript

let day = 3;
switch (day) {
  case 1: console.log("Monday"); break;
  case 2: console.log("Tuesday"); break;
  case 3: console.log("Wednesday"); break;
  default: console.log("Invalid day");
}
7. Looping statements with example:
javascript

// for loop
for (let i = 1; i <= 5; i++) {
  console.log(i);
}

// while loop
let j = 1;
while (j <= 5) {
  console.log(j);
  j++;
}

// do-while loop
let k = 1;
do {
  console.log(k);
  k++;
} while (k <= 5);
8. Loop control statements with example:
javascript

// break
for (let i = 1; i <= 5; i++) {
  if (i === 3) break;
  console.log(i); // prints 1, 2
}

// continue
for (let i = 1; i <= 5; i++) {
  if (i === 3) continue;
  console.log(i); // skips 3
}
